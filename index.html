const FOLDER_ID = '1H9w8hjb-6SdBBsN1xGN1LEWPT0bnsZT6';

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    const folder = DriveApp.getFolderById(payload.folderId || FOLDER_ID);

    // 方案 A：雙向同步刪除雲端硬碟照片
    if (payload.action === 'delete' && payload.driveFileId) {
      try {
        DriveApp.getFileById(payload.driveFileId).setTrashed(true);
      } catch (delErr) {}
      return ContentService.createTextOutput(JSON.stringify({ status: 'ok', deleted: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // 接收高畫質 Base64 照片並寫入指定 Google Drive 資料夾
    const base64Data = payload.dataUrl.split(',')[1];
    const contentType = payload.mimeType || 'image/jpeg';
    const blob = Utilities.newBlob(Utilities.base64Decode(base64Data), contentType, payload.fileName || ('asset_' + Date.now() + '.jpg'));
    const file = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

    const fileId = file.getId();
    return ContentService.createTextOutput(JSON.stringify({
      status: 'ok',
      driveFileId: fileId,
      driveUrl: 'https://drive.google.com/uc?export=view&id=' + fileId,
      viewUrl: 'https://drive.google.com/file/d/' + fileId + '/view',
      thumbUrl: 'https://drive.google.com/thumbnail?id=' + fileId + '&sz=w1600'
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
